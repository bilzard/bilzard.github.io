* LCD 1602 Display

## Setup

D4~D9端子を使ってディスプレイに移動平均と移動標準偏差を表示する。

![](/images/diode_exp_ex/07_exp_setup.jpg)

右側はディスプレイ表示回路。左側は計測対象の系。

## 実験方法

20msの間に5msずつdischargeし、それらの平均を計算する。ノイズが商用電源の50Hzであることから、その周期20ms間に1位相が経過する。
この間のノイズの影響を平均化することが目的。コードは付録Aに記載。

## 実験結果

condition | (1) Light: Off \[V\] | (2) Light: On \[V\] | (2) - (1) \[V\]
:--: | :-- | --: | --: | --:
ノイズ源なし | -0.050 (+-0.001) | 0.387 (+-0.002) | 0.437
OA tap接続時 | -0.030 (+-0.006) | 0.378 (+-0.004) | 0.408
OA tap, AC adapter接続時 | 0.052 (+-0.014) | 0.400 (+-0.333) | 0.348
コンセント→USBハブ給電時 | 0.0 (+-0.19) | 0.35 (+-0.25) | 0.35

ノイズが大きいほど、Offの時のdiffが小さく、光源を点灯・消灯時のdiffの変化が小さくなる傾向がみられた。
→ **ON/OFFの信号のレベル差がノイズの紛れやすくなる**。

## 議論

* 前回の実験の時の5Vのノイズが再現できなかった（バッテリーの残量に依存する？）→ バッテリーの残量を減らしてから再実験する
* USBハブからの給電ではノイズが非常に大きく、ON/OFFの信号のレベル差が読み取りづらくなる。 → 閾値で判定する場合はノイズの強さで変わるので閾値を環境に合わせて調整しないといけなくなる
* ノイズ源の周波数（50Hz）を決めうちにした対策であり、ノイズ源の周波数が変わった場合に使えなくなる。 → 市販のリモコンのように高周波のパルス変調をかけてhigh pass filterを通すとか？

### B. ソースコード

```c
#include <LiquidCrystal.h>
#include <math.h>

// LCDピン設定
LiquidCrystal lcd(4, 5, 6, 7, 8, 9);

const int PIN_SENSOR = A0;
const int PIN_REF    = A1;
int sampleWindow = 20;

// --- 統計計算用設定 ---
const int WINDOW_SIZE = 30;     // 統計を取るサンプル数
float diffHistory[WINDOW_SIZE];
int historyIndex = 0;
bool bufferFull = false;

// --- LCD表示更新用 ---
unsigned long lastDisplayTime = 0;
const int displayInterval = 250;

void setup() {
  Serial.begin(9600);
  lcd.begin(16, 2);

  // バッファ初期化
  for (int i = 0; i < WINDOW_SIZE; i++) diffHistory[i] = 0.0;

  lcd.print("Stats Mode");
  delay(1000);
  lcd.clear();
}

void loop() {
  // -------------------------------------------------------
  // 1. 同時に充電 (元の行動・タイミングを維持)
  // -------------------------------------------------------
  pinMode(PIN_SENSOR, OUTPUT);
  pinMode(PIN_REF, OUTPUT);
  digitalWrite(PIN_SENSOR, HIGH);
  digitalWrite(PIN_REF, HIGH);
  delay(2);

  // -------------------------------------------------------
  // 2. 同時に放電開始
  // -------------------------------------------------------
  pinMode(PIN_SENSOR, INPUT);
  digitalWrite(PIN_SENSOR, LOW);
  pinMode(PIN_REF, INPUT);
  digitalWrite(PIN_REF, LOW);

  // -------------------------------------------------------
  // 3. 勾配（放電）を待つ
  // -------------------------------------------------------
  delay(sampleWindow);

  // -------------------------------------------------------
  // 4. サンプリング
  // -------------------------------------------------------
  int valSensor = analogRead(PIN_SENSOR);
  int valRef    = analogRead(PIN_REF);

  // 電圧換算
  float volSensor = 5.0 * (1023.0 - valSensor) / 1023.0;
  float volRef    = 5.0 * (1023.0 - valRef) / 1023.0;
  float currentDiff = volSensor - volRef;

  // -------------------------------------------------------
  // 5. プロッタ出力 (以前の通り2信号とその差を表示)
  // -------------------------------------------------------
  Serial.print(volSensor, 4);
  Serial.print(",");
  Serial.print(volRef, 4);
  Serial.print(",");
  Serial.println(currentDiff, 4);

  // -------------------------------------------------------
  // 6. 統計バッファの更新
  // -------------------------------------------------------
  diffHistory[historyIndex] = currentDiff;
  historyIndex = (historyIndex + 1) % WINDOW_SIZE;
  if (historyIndex == 0) bufferFull = true;

  // -------------------------------------------------------
  // 7. LCD表示の更新 (指定間隔ごと)
  // -------------------------------------------------------
  if (millis() - lastDisplayTime >= displayInterval) {

    // 平均と標準偏差の計算
    float sum = 0;
    float sumSq = 0;
    int count = bufferFull ? WINDOW_SIZE : historyIndex;

    if (count > 0) {
      for (int i = 0; i < count; i++) {
        sum += diffHistory[i];
        sumSq += pow(diffHistory[i], 2);
      }
      float movingAvg = sum / count;
      float variance = (sumSq / count) - pow(movingAvg, 2);
      if (variance < 0) variance = 0;
      float movingStdDev = sqrt(variance);

      // 表示実行
      lcd.setCursor(0, 0);
      lcd.print("AVG : ");
      lcd.print(movingAvg, 3);
      lcd.print("V  ");

      lcd.setCursor(0, 1);
      lcd.print("S.D.: ");
      lcd.print(movingStdDev, 4); // 標準偏差
      lcd.print("    ");
    }

    lastDisplayTime = millis();
  }
}
```
